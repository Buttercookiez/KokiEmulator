cmake_minimum_required(VERSION 3.22)
project(koki_nds LANGUAGES C CXX)
set(BUILD_QT_SDL OFF CACHE BOOL "" FORCE)
set(ENABLE_JIT ON CACHE BOOL "" FORCE)
set(ENABLE_GDBSTUB OFF CACHE BOOL "" FORCE)
set(ENABLE_OGLRENDERER OFF CACHE BOOL "" FORCE)
set(ENABLE_LTO_RELEASE OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
if(NOT KOKI_NDS_SOURCE)
  get_filename_component(KOKI_NDS_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../.cache/native-nds/core-build" ABSOLUTE)
endif()
add_subdirectory("${KOKI_NDS_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/melonds")
target_compile_definitions(core PRIVATE KOKI_NDS_BOUNDED_MEMORY=1)
add_library(koki_nds SHARED koki_nds.cpp platform.cpp)
target_link_libraries(koki_nds PRIVATE core android log z)
target_compile_options(koki_nds PRIVATE -O3 -fwrapv)
target_link_options(koki_nds PRIVATE -Wl,-z,max-page-size=16384)
target_link_libraries(koki_nds PRIVATE c++_static c++abi)
